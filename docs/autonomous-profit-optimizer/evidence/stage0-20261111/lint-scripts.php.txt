<?php
$files = ['artisan','public/index.php'];
foreach (new RecursiveIteratorIterator(new RecursiveDirectoryIterator('scripts', FilesystemIterator::SKIP_DOTS)) as $file) {
 if ($file->getExtension() === 'php') $files[]=$file->getPathname();
}
$failures=0;
foreach ($files as $file) {
 $output=[]; exec(escapeshellarg(PHP_BINARY).' -l '.escapeshellarg($file).' 2>&1',$output,$code);
 if ($code!==0) { echo implode("\n",$output)."\n"; $failures++; }
}
echo count($files)." additional PHP scripts/entrypoints; failures: ".$failures."\n";
exit($failures===0?0:1);
